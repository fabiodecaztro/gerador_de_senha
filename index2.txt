        function generatePasswords() {
            // Obter valores dos campos
            const passwordCount = parseInt(document.getElementById('passwordCount').value) || 1;
            const passwordLength = parseInt(document.getElementById('passwordLength').value) || 12;
            
            // Obter tipos de caracteres selecionados
            const includeNumbers = document.getElementById('includeNumbers').checked;
            const includeSpecial = document.getElementById('includeSpecial').checked;
            const includeUppercase = document.getElementById('includeUppercase').checked;
            const includeLowercase = document.getElementById('includeLowercase').checked;
            
            const resultArea = document.getElementById('resultArea');
            
            // VALIDAÇÕES PERMANECEM AQUI PARA GARANTIR QUE A LÓGICA DE GERAÇÃO DE SENHAS FUNCIONE CORRETAMENTE
            // E PARA EXIBIR MENSAGENS DE ERRO CLARAS AO USUÁRIO APÓS TENTAR GERAR
            if (passwordLength < 4 || passwordLength > 50) {
                resultArea.innerHTML = '<div class="error-message">A quantidade de caracteres deve ser entre 4 e 50.</div>';
                return;
            }
            
            if (passwordCount < 1 || passwordCount > 200) {
                resultArea.innerHTML = '<div class="error-message">A quantidade de senhas deve ser entre 1 e 200.</div>';
                return;
            }

            if (!includeNumbers && !includeSpecial && !includeUppercase && !includeLowercase) {
                resultArea.innerHTML = '<div class="error-message">Por favor, selecione pelo menos um tipo de caractere.</div>';
                return;
            }
            
            // Definir conjuntos de caracteres
            const numbers = '0123456789';
            const special = '!@#$%^&*';
            const uppercase = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
            const lowercase = 'abcdefghijklmnopqrstuvwxyz';
            
            // Criar array de tipos selecionados
            const selectedTypes = [];
            if (includeNumbers) selectedTypes.push({ chars: numbers, name: 'numbers' });
            if (includeSpecial) selectedTypes.push({ chars: special, name: 'special' });
            if (includeUppercase) selectedTypes.push({ chars: uppercase, name: 'uppercase' });
            if (includeLowercase) selectedTypes.push({ chars: lowercase, name: 'lowercase' });
            
            // Gerar senhas
            const passwords = [];
            
            for (let i = 0; i < passwordCount; i++) {
                passwords.push(generateSinglePassword(passwordLength, selectedTypes));
            }
            
            // Exibir resultados
            displayPasswords(passwords);
        }
        
        function generateSinglePassword(length, selectedTypes) {
            let password = '';
            const allChars = selectedTypes.map(type => type.chars).join('');
            
            // Garantir que pelo menos um caractere de cada tipo selecionado seja incluído
            const guaranteedChars = [];
            selectedTypes.forEach(type => {
                const randomChar = type.chars[Math.floor(Math.random() * type.chars.length)];
                guaranteedChars.push(randomChar);
            });
            
            // Preencher o restante da senha
            const remainingLength = length - guaranteedChars.length;
            const randomChars = [];
            
            for (let i = 0; i < remainingLength; i++) {
                const randomChar = allChars[Math.floor(Math.random() * allChars.length)];
                randomChars.push(randomChar);
            }
            
            // Combinar e embaralhar todos os caracteres
            const allPasswordChars = [...guaranteedChars, ...randomChars];
            
            // Embaralhar usando algoritmo Fisher-Yates
            for (let i = allPasswordChars.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [allPasswordChars[i], allPasswordChars[j]] = [allPasswordChars[j], allPasswordChars[i]];
            }
            
            return allPasswordChars.join('');
        }
        
        function displayPasswords(passwords) {
            const resultArea = document.getElementById('resultArea');
            
            let html = `<div class="success-message">✓ ${passwords.length} senha${passwords.length > 1 ? 's' : ''} gerada${passwords.length > 1 ? 's' : ''} com sucesso!</div>`;
            
            passwords.forEach((password, index) => {
                html += `<div class="password-item">${password}</div>`;
            });
            
            resultArea.innerHTML = html;
        }
        
        document.addEventListener('DOMContentLoaded', function() {
            document.addEventListener('keypress', function(e) {
                if (e.key === 'Enter') {
                    generatePasswords();
                }
